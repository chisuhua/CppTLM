// 设备注册表 (per Oracle Q6 多卡并发)
namespace {
  std::mutex registry_mu_;
  std::unordered_map<uint32_t, DGpuBoard*> registry_;
  std::atomic<uint32_t> next_dev_id_{1};
  inline DGpuBoard* lookup(uint32_t dev_id) {
    auto it = registry_.find(dev_id);
    return it != registry_.end() ? it->second : nullptr;
  }
}

// 销毁顺序 (per design §4): 锁内 erase, 解锁后 delete
void cpptlm_emulator_destroy(cpptlm_emulator_t* emu) {
  std::unique_lock<std::mutex> lock(registry_mu_);
  registry_.erase(emu->dev_id);
  lock.unlock();
  delete emu->board;  // 解锁后析构,防 sim_thread 持锁死锁
}

// get_version 返回字面量 (per §D6.2)
const char* cpptlm_emulator_get_version() {
  return "v1.0-dgpu-v0";
}
